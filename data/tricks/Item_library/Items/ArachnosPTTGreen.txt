	Multiply1Scale 10 10

	BumpMap1Scale 1 1

	MaskScale 1 1

	Base2 none

	Base2Scale 1 1

	Multiply2 goth_reflect_03.tga

	Multiply2Scale 1 -1
	Multiply2reflect 1

	DualColor2 texture_name.tga
	Color3 0 0 0
	Color4 0 0 0
	DualColor2Scroll 0 0
	DualColor2Scale 1 1

	BumpMap2 none //arac_machinery_t_01_bump.tga

	BumpMap2Scale 1 1

	SpecularColor1 170 245 190 
	SpecularColor2 125 150 250

	AmbientScale 2

	MaxAddGlow 65

		UseFallback 1
		Base AracSwap_machinery_t_01-green.tga
		Blend none
		Bumpmap arac_machinery_t_01_bump.tga
		BlendType Multiply
		ScaleST0 1 1
		ScaleST1 1 1
		DiffuseScale 1.5
		AmbientScale 1.5
	End

End
