#	BlendType AddGlow

# new shader

Texture freedom_logo01
	Base1 freedom_logo01.tga
	Base1Scroll 0 0
	Base1ScrollType Normal
	Base1Scale 1 1

	Multiply1 none
	Multiply1Scroll 0 0
	Multiply1ScrollType Normal
	Multiply1Scale .1 .1
	Multiply1Reflect 0 // Optional flag indicating this is a spheremap

	DualColor1 none
	DualColor1Scroll 0 0
	DualColor1ScrollType Normal
	DualColor1Scale 1 1

	BumpMap1 freedom_logo01_bump.tga
	BumpMap1Scroll 0 0
	BumpMap1ScrollType Normal
	BumpMap1Scale 1 1

	AddGlow1 freedom_logo01_glow.tga
	AddGlow1Scroll 0 0
	AddGlow1ScrollType Normal
	AddGlow1Scale 1 1

	Mask black
	MaskScroll 0 0
	MaskScrollType Normal
	MaskScale 1 1
	MaskWeight .2
	AlphaMask 0	// Optional flag indicating to use the ALPHA channel
			// of the Mask texture instead of all 4 channels

	Base2 none
	Base2Scroll 0 0
	Base2ScrollType Normal
	Base2Scale 1 1

	Multiply2 universal_window_reflect5c.tga
	Multiply2Scroll 0 -0.00009
	Multiply2ScrollType Normal
	Multiply2Scale 2 2
	Multiply2Reflect 1 // Optional flag indicating this is a spheremap

	DualColor2 none
	DualColor2Scroll 0 0
	DualColor2ScrollType Normal
	DualColor2Scale 1 1

	BumpMap2 none
	BumpMap2Scroll 0 0
	BumpMap2ScrollType Normal
	BumpMap2Scale 1 1

	SpecularColor1 255 255 255
	SpecularColor2 0 0 0 // BLACK means inherit color from SpecularColor1
	SpecularExponent1 32 
	SpecularExponent2 8

	DiffuseScale 1.0
	AmbientScale 2.0

	Flags none // NoRandomAddGlow FullBright
	MinAddGlow 0
	MaxAddGlow 128

	Color3 0 0 0 // Color tint for DualColor2, black inherits from colors 1 and 2
	Color4 0 0 0 

	Fallback
		UseFallback 0
		Base none
		Blend none
		Bumpmap none
		BlendType AddGlow // DualColor, Multiply, AddGlow, or AlphaDetail
		ScaleST0 1 1
		ScaleST1 1 1
		DiffuseScale 1.0
		AmbientScale 1.0
	End
End
