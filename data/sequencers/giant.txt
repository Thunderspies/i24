###### Motion  ###########################################################
##  Note: Giant uses both motion here and from the motion.inc files	##
##	You Must name each move with a unique prefix "G_" and set the	##
##	priority one higher than the motion.inc file			##
##########################################################################

###################
#### run Forward ##
####################

Move G_RunPre
	Requires	FORWARD
	Priority        2
	Interrupts	<Ready>  <PostMoveCycle> <Block> <HitIRQ> RH_RunPre RH_RunCycle
	Member		<BasicMove> <DeathIRQ> <AttackIRQ>
	Flags		ReqInputs MoveScale
	NextMove        G_RunCycle
	Interpolate	8

	Type		male
	  Anim          huge\giant_club_prerun 1
	  PlayFx        FEET\MALERIGHT.FX 10
	  Scale		1.2
	   MoveRate     0.7
	TEnd
MEnd

Move G_RunCycle
	Requires	FORWARD
	Priority        2
	Member		<BasicMove> <DeathIRQ> <AttackIRQ>
	Flags		Cycle MoveScale
	NextMove	G_RunPost
	Interpolate	8

	Type		male
	  Anim          huge\giant_head_run 1
	  PlayFx        FEET\MALELEFT.FX 10
	  PlayFx        FEET\MALERIGHT.FX 21
	  Scale		1.2
	  MoveRate      0.7
	TEnd
MEnd

Move G_RunPost
	Interrupts	RunPost
	Include sequencers\Members\Readies.inc
	NextMove        Ready
	Interpolate	8

	Type		male
	  Anim          huge\giant_club_postrun 1
	  PlayFx	FEET\LAND3.FX:LAND  3
	  PlayFx	FEET\MALERIGHT.FX:FOOTFALLRIGHT 12
	  Scale		1.5
	TEnd
MEnd
